/** 返回呼吸缩放系数，围绕 1.0 上下浮动 amplitude。 */
export function breatheScale(
  elapsedMs: number,
  periodMs: number,
  amplitude: number,
): number {
  const phase = (elapsedMs % periodMs) / periodMs;
  return 1 + amplitude * Math.sin(phase * Math.PI * 2);
}
```

export type PetActivity = "sleep" | "idle" | "active";

const TARGET_FPS: Record<PetActivity, number> = {
  sleep: 4,
  idle: 12,
  active: 30,
};

export function frameIntervalMs(activity: PetActivity): number {
  return 1000 / TARGET_FPS[activity];
}

export function shouldRender(
  nowMs: number,
  lastRenderMs: number,
  activity: PetActivity,
): boolean {
  return nowMs - lastRenderMs >= frameIntervalMs(activity);
}

import { isInsideBox, type Box } from "./hit-test";

export interface DragState {
  dragging: boolean;
  offsetX: number;
  offsetY: number;
}

export interface Pointer {
  px: number;
  py: number;
}

export function createDragState(): DragState {
  return { dragging: false, offsetX: 0, offsetY: 0 };
}

export function onPointerDown(_s: DragState, p: Pointer, body: Box): DragState {
  if (!isInsideBox(p.px, p.py, body)) return createDragState();
  return { dragging: true, offsetX: p.px - body.x, offsetY: p.py - body.y };
}

export function onPointerMove(
  s: DragState,
  p: Pointer,
): { x: number; y: number } | null {
  if (!s.dragging) return null;
  return { x: p.px - s.offsetX, y: p.py - s.offsetY };
}

export function onPointerUp(_s: DragState): DragState {
  return createDragState();
}
```

**Step 1: 写 `src/pet.ts`**

注意 `SIZE_STEPS` 是整数倍档位（设计要求），以及 `dithering` 辉光 —— alpha 只取 0 或 1，避免形成挡鼠标的「隐形墙」。

```ts
import { breatheScale } from "./anim/breathe";
import { frameIntervalMs, shouldRender, type PetActivity } from "./anim/frame-budget";
import {
  createDragState,
  onPointerDown,
  onPointerMove,
  onPointerUp,
  type DragState,
} from "./interact/drag";
import { isInsideBox, type Box } from "./interact/hit-test";

/** 像素艺术必须整数倍缩放，否则糊。基础格 48px。 */
const BASE_CELL = 48;
export const SIZE_STEPS = [2, 3, 4, 5] as const;

const BODY_COLOR = "#7cf5c4";
const GLOW_COLOR = "#7cf5c4";
const POKE_COLOR = "#ffe066";

export class Pet {
  private x = 200;
  private y = 200;
  private scaleStep = 3;
  private activity: PetActivity = "idle";
  private drag: DragState = createDragState();
  private lastRenderMs = 0;
  private pokeUntilMs = 0;
  private startMs = performance.now();

  constructor(
    private readonly canvas: HTMLCanvasElement,
    private readonly ctx: CanvasRenderingContext2D,
  ) {}

  private get side(): number {
    return BASE_CELL * SIZE_STEPS[this.scaleStep - 2];
  }

  get body(): Box {
    return { x: this.x, y: this.y, w: this.side, h: this.side };
  }

  private get poked(): boolean {
    return performance.now() < this.pokeUntilMs;
  }

  pointerDown(px: number, py: number): boolean {
    const hit = isInsideBox(px, py, this.body);
    if (!hit) return false;
    this.drag = onPointerDown(this.drag, { px, py }, this.body);
    this.pokeUntilMs = performance.now() + 400;
    this.activity = "active";
    return true;
  }

  pointerMove(px: number, py: number): void {
    const pos = onPointerMove(this.drag, { px, py });
    if (!pos) return;
    this.x = pos.x;
    this.y = pos.y;
  }

  pointerUp(): void {
    this.drag = onPointerUp(this.drag);
    this.activity = "idle";
  }

  /** 由 rAF 驱动；返回是否真的绘制了（用于 CPU 观察）。 */
  tick(nowMs: number): boolean {
    const activity = this.poked || this.drag.dragging ? "active" : this.activity;
    if (!shouldRender(nowMs, this.lastRenderMs, activity)) return false;
    this.lastRenderMs = nowMs;
    this.draw(nowMs);
    return true;
  }

  private draw(nowMs: number): void {
    const { ctx } = this;
    ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);

    const scale = breatheScale(nowMs - this.startMs, 2400, 0.04);
    const side = Math.round(this.side * scale);
    // 缩放围绕中心，避免呼吸时看起来在往右下漂
    const px = Math.round(this.x + (this.side - side) / 2);
    const py = Math.round(this.y + (this.side - side) / 2);

    this.drawDitherGlow(px, py, side);

    ctx.fillStyle = this.poked ? POKE_COLOR : BODY_COLOR;
    ctx.fillRect(px, py, side, side);

    // 两只眼睛，确认「有朝向」的观感
    const eye = Math.max(2, Math.round(side / 12));
    ctx.fillStyle = "#0b1020";
    ctx.fillRect(px + Math.round(side * 0.28), py + Math.round(side * 0.32), eye, eye);
    ctx.fillRect(px + Math.round(side * 0.62), py + Math.round(side * 0.32), eye, eye);
  }

  /**
   * dithering 辉光：棋盘点阵，alpha 只有 0 或 1。
   * 不用 CSS bloom / shadowBlur —— 那会产生大片低 alpha 像素，
   * 在 macOS 上形成挡住鼠标的「隐形墙」。
   */
  private drawDitherGlow(px: number, py: number, side: number): void {
    const { ctx } = this;
    const cell = Math.max(2, Math.round(side / 24));
    ctx.fillStyle = GLOW_COLOR;
    for (let ring = 1; ring <= 2; ring++) {
      const inset = ring * cell;
      const x0 = px - inset;
      const y0 = py - inset;
      const w = side + inset * 2;
      const h = side + inset * 2;
      for (let i = 0; i < w; i += cell) {
        for (let j = 0; j < h; j += cell) {
          const onEdge = i < cell || j < cell || i >= w - cell || j >= h - cell;
          if (!onEdge) continue;
          // 棋盘取点：ring 越外，点越稀
          if (((i / cell + j / cell + ring) | 0) % (ring + 1) !== 0) continue;
          ctx.fillRect(x0 + i, y0 + j, cell, cell);
        }
      }
    }
  }

  resize(w: number, h: number): void {
    this.canvas.width = w;
    this.canvas.height = h;
    this.lastRenderMs = 0;
  }

  cycleSize(): void {
    this.scaleStep = this.scaleStep >= 5 ? 2 : this.scaleStep + 1;
  }

  setActivity(a: PetActivity): void {
    this.activity = a;
  }

  get debugIntervalMs(): number {
    return frameIntervalMs(this.activity);
  }
}
```

**实施记录：对本任务原始代码的两处修正**

原计划的代码有两个问题，实现时已修正，理由如下：

1. **尺寸档位的表示方式**。原代码用 `scaleStep = 3` 表示倍数，取值时 `SIZE_STEPS[this.scaleStep - 2]`，循环时 `scaleStep >= 5 ? 2 : scaleStep + 1` —— 倍数与下标混用，边界值 `2`/`5` 硬编码在三个地方，改档位列表就会错。改为直接持有下标 `sizeIndex`，循环用 `(sizeIndex + 1) % SIZE_STEPS.length`，档位列表成为唯一事实来源。

2. **交互会污染活跃度状态**（真实 bug）。原代码在 `pointerDown` 里写 `this.activity = "active"`、在 `pointerUp` 里写 `this.activity = "idle"`。后果是：宠物本处于 `sleep`，你拖它一下，松手后它变成 `idle` 而非回到 `sleep` —— 用户设置被交互静默覆盖。M2 接入自动状态感知后，这会表现为「摸一下宠物，它就忘了你其实已经离开了」。改为用派生方法 `effectiveActivity()` 在渲染时临时提档，不写回 `this.activity`。

另外去掉了 dither 循环里多余的 `| 0`：`i` 与 `j` 都是 `cell` 的整数倍，`i / cell` 必为整数。

**追加测试（原计划未要求）**

原计划 Task 5 只安排浏览器目视验证。但「整数倍缩放」是设计文档定义的生死线约束，而目视**看不出它有没有被破坏** —— 一旦有人往 `SIZE_STEPS` 加个 `2.5`，画面只是轻微发虚，很难察觉，却已破坏整套像素质感。

因此补充 `tests/pet.test.ts`（12 个测试），覆盖：边长恒为基础格整数倍、档位循环不越界、宠物恒为正方形、拖动偏移不跳位、松手后不再跟随、睡眠态跳过过密 tick、拖动中强制提到 active、resize 后强制重绘。

已验证这些测试能真正抓 bug：向 `SIZE_STEPS` 加入 `2.5` 会让整数倍测试失败；移除 drag 的 offset 记录会让「不跳位」测试失败；移除 `resize` 中的 `lastRenderMs = 0` 会让重绘测试失败。

```rust
use tauri::{AppHandle, Manager, PhysicalPosition, PhysicalSize};

/// 把宠物窗口铺满主显示器，并做平台特化处理。
pub fn setup_pet_window(app: &AppHandle) -> tauri::Result<()> {
    let win = app
        .get_webview_window("pet")
        .expect("window with label 'pet' must exist in tauri.conf.json");

    if let Some(monitor) = win.primary_monitor()? {
        let size = monitor.size();
        let pos = monitor.position();
        win.set_size(PhysicalSize::new(size.width, size.height))?;
        win.set_position(PhysicalPosition::new(pos.x, pos.y))?;
    }

    #[cfg(target_os = "macos")]
    macos::convert_to_pet_panel(&win);

    win.show()?;
    Ok(())
}

#[cfg(target_os = "macos")]
mod macos {
    use tauri::WebviewWindow;
    use tauri_nspanel::{
        tauri_panel, CollectionBehavior, PanelLevel, StyleMask, WebviewWindowExt,
    };

    tauri_panel! {
        panel!(PetPanel {
            config: {
                is_floating_panel: true,
                can_become_key_window: true,
                can_become_main_window: false
            }
        })
    }

    /// 关键配方，详见 docs/plans/2026-08-29-vibe-pet-design.md 3.3。
    /// 全程用 objc2::exception::catch 兜底：msg_send! 遇 Obj-C 异常会 panic，
    /// 拦住之后最坏只是退化成普通 NSWindow，不会 SIGABRT。
    pub fn convert_to_pet_panel(win: &WebviewWindow) {
        use std::panic::AssertUnwindSafe;

        let result = tauri_nspanel::objc2::exception::catch(AssertUnwindSafe(|| {
            let panel = match win.to_panel::<PetPanel>() {
                Ok(p) => p,
                Err(e) => {
                    eprintln!("[pet] to_panel failed: {e:?}");
                    return;
                }
            };

            // Dock level + nonactivating 才会被 AppKit 路由进全屏辅助层
            panel.set_level(PanelLevel::Dock.value());

            // 生死线：点宠物不抢走编辑器焦点
            panel.set_style_mask(StyleMask::empty().nonactivating_panel().into());

            // 跟随所有 Space，可浮在全屏应用之上
            panel.set_collection_behavior(
                CollectionBehavior::new()
                    .stationary()
                    .can_join_all_spaces()
                    .full_screen_auxiliary()
                    .into(),
            );

            // nonactivating 只管焦点，不管 resign-active 时的自动隐藏。
            // 不设这行会出现「要点一下宠物它才回来」的延迟。
            panel.set_hides_on_deactivate(false);

            // 防 Touch Bar KVO 崩溃：to_panel 换类发生在 Touch Bar finder
            // 已对原 NSWindow 注册观察者之后，注销时会抛 NSRangeException。
            let _: () = unsafe {
                tauri_nspanel::objc2::msg_send![
                    panel.as_panel(),
                    setAutorecalculatesTouchBar: false
                ]
            };
        }));

        if result.is_err() {
            eprintln!("[pet] panel setup raised an Obj-C exception; running as plain NSWindow");
        }
    }
}
```

use tauri::{
    menu::{Menu, MenuItem},
    tray::TrayIconBuilder,
    App, Manager,
};

pub fn setup_tray(app: &App) -> tauri::Result<()> {
    let toggle = MenuItem::with_id(app, "toggle", "显示 / 隐藏", true, None::<&str>)?;
    let quit = MenuItem::with_id(app, "quit", "退出", true, None::<&str>)?;
    let menu = Menu::with_items(app, &[&toggle, &quit])?;

    TrayIconBuilder::with_id("pet-tray")
        .tooltip("Vibe Pet")
        .menu(&menu)
        .show_menu_on_left_click(true)
        .on_menu_event(|app, event| match event.id.as_ref() {
            "toggle" => {
                if let Some(win) = app.get_webview_window("pet") {
                    let visible = win.is_visible().unwrap_or(false);
                    let _ = if visible { win.hide() } else { win.show() };
                }
            }
            "quit" => app.exit(0),
            _ => {}
        })
        .build(app)?;

    Ok(())
}
